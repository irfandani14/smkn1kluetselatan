Skrip SQL Migrasi Database

Buka Supabase Dashboard > Proyek Anda > Menu SQL Editor > Tempel skrip ini dan klik RUN.

-- ==============================================================================
-- SISTEM ABSENSI WAJAH REALTIME - SKEMA DATABASE SUPABASE
-- ==============================================================================

CREATE TABLE IF NOT EXISTS public.students (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nisn TEXT UNIQUE NOT NULL,
    name TEXT NOT NULL,
    class_name TEXT NOT NULL,
    face_descriptor JSONB,
    photo_url TEXT,
    created_at TIMESTAMPTZ DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_students_nisn ON public.students(nisn);
CREATE INDEX IF NOT EXISTS idx_students_class ON public.students(class_name);

CREATE TABLE IF NOT EXISTS public.attendances (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    student_id UUID NOT NULL REFERENCES public.students(id) ON DELETE CASCADE,
    class_name TEXT NOT NULL,
    date DATE NOT NULL DEFAULT CURRENT_DATE,
    time TIME NOT NULL DEFAULT CURRENT_TIME,
    status TEXT NOT NULL DEFAULT 'Hadir',
    created_at TIMESTAMPTZ DEFAULT now(),
    CONSTRAINT unique_student_daily_attendance UNIQUE (student_id, date)
);

CREATE INDEX IF NOT EXISTS idx_attendances_date ON public.attendances(date);
CREATE INDEX IF NOT EXISTS idx_attendances_student ON public.attendances(student_id);

ALTER TABLE public.students ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.attendances ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Akses Penuh Siswa untuk Aplikasi Sekolah" ON public.students;
CREATE POLICY "Akses Penuh Siswa untuk Aplikasi Sekolah"
    ON public.students FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Akses Penuh Presensi untuk Aplikasi Sekolah" ON public.attendances;
CREATE POLICY "Akses Penuh Presensi untuk Aplikasi Sekolah"
    ON public.attendances FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

ALTER PUBLICATION supabase_realtime ADD TABLE public.attendances;
